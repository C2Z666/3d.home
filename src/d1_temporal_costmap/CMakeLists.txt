cmake_minimum_required(VERSION 3.16)
project(d1_temporal_costmap)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
add_library(d1_temporal_costmap SHARED src/temporal_voxel_layer.cpp)
target_compile_features(d1_temporal_costmap PUBLIC cxx_std_17)
target_compile_options(d1_temporal_costmap PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(d1_temporal_costmap nav2_costmap_2d pluginlib)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS d1_temporal_costmap LIBRARY DESTINATION lib)
ament_package()
